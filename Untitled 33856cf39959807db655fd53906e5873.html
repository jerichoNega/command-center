setSyncKey(key);
loadRemoteData(remoteData);

} catch {}

          {[["focus","25 min focus","#c8a96e"],["break","5 min break","#7eb8a4"]].map(([mode,label,color]) => (

    )}

  </div>

  {/* Footer sync key */}

      style={{background:"none",border:"none",fontFamily:"'DM Mono',monospace",fontSize:"9px",color:"#2a2a30",letterSpacing:"0.08em",cursor:"pointer"}}>
